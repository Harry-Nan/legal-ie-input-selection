aantal zendingen :

max. 1;

aard van de stof :

uranium-238;

gewicht U-totaal per verpakking :

max. 8300 kg;

hoeveelheid verpakking per zending :

max. 1 verpakking met in totaal max. 8300 kg uranium (in de vorm van besmet staalschroot) zich bevindend in 20ft ISO container;

verrijkingsgraad :

0%;

fysische toestand :

vast;

chemische toestand :

metaal;

totale activiteit per verpakking :

max. 91 megabecquerel (MBq).

afkomstig van :

Cronimet Holland B.V. te Moerdijk;

bestemd voor :

Cronimet Legierungen Dortmund GmbH te Dortmund (Duitsland);

vervoerder over de weg :

de vergunninghouder;

type verpakking

:

IP 2 (type 20ft ISO container);

stralingscategorie :

I-wit;

transportindex :

max. 1 per verpakking.
